Form Validator

// The Prompt

Create a form with 4 inputs and a submit button. When clicked, loop through the inputs. If empty, turn border red. If filled, turn border green.

// The Code

                  
            function regBtn(element) {
                // Stop the form from refreshing the page first!
                element.preventDefault();

                // Grab all inputs inside the form
                let myInputs = element.target.parentElement.querySelectorAll("input");

                for (let i = 0; i < myInputs.length; i++) {
                    if (myInputs[i].value == "") {
                        myInputs[i].style.border = "2px solid maroon";
                    } else {
                        myInputs[i].style.border = "2px solid lime";
                    }
                }
            }
