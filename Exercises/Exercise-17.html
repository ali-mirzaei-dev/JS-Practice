// The Prompt

Create a Frequently Asked Questions component. When a button is clicked, toggle the visibility of the answer paragraph and change the button's text.

// The Code

                  
            function show(element) {
                let myP = element.parentElement.querySelector("p");

                if (element.innerText == "Show Answer") {
                    element.innerText = "Hide Answer";
                } else {
                    element.innerText = "Show Answer";
                }

                if (myP.style.display == "flex") {
                    myP.style.display = "none";
                } else {
                    myP.style.display = "flex";
                }
            }
