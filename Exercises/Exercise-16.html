Dynamic Card Builder

// The Prompt

Create an empty div and a button. Every time the button is clicked, use .innerHTML to inject a new HTML card into the div. Add a close button to remove the card.

// The Code

                  
            const myId = document.getElementById("myDiv");

            function contentAdd() {
                
                let card = "<div class='bg-white/5 border border-white/10 p-5 rounded-lg flex justify-between items-center'>";
                card += "<div>";
                card += "<h3 class='text-xl text-accent font-bold mb-2'>New Card</h3>";
                card += "<p class='text-text-secondary text-sm'>Generated dynamically via JS!</p>";
                card += "</div>";
                card += "<button onclick='closer(this)' class='px-2.5 py-1 text-xs text-red-400 bg-red-500/20 border border-red-500/40
                 rounded hover:bg-red-500 hover:text-white transition-colors'>Close</button>";
                card += "</div>";

                myId.innerHTML += card;
            }

            function closer(element) {
                element.parentElement.remove();
            }
