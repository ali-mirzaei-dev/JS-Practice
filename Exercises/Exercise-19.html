Grade Calculator

// The Prompt

Create an input field and a calculate button. When clicked, read the input value and use if/else if/else to print a letter grade (A-F) into an empty div based on the number.

// The Code

                          
                const myBtn = document.querySelector("form>button");
                const myOutPut = document.getElementById("outPut");

                function gridCalc(element) {
                    element.preventDefault();

                    let score = element.target.parentElement.querySelector("input").value;

                    if (score >= 90 && score <= 100) {
                        myOutPut.innerHTML = "<h1>You got an A</h1>";
                        myOutPut.style.border = "2px solid lime";
                    } else if (score >= 80 && score < 90) {
                        myOutPut.innerHTML = "<h1>You got a B</h1>";
                        myOutPut.style.border = "2px solid blue";
                    } else if (score >= 70 && score < 80) {
                        myOutPut.innerHTML = "<h1>You got a C</h1>";
                        myOutPut.style.border = "2px solid yellow";
                    } else if (score >= 60 && score < 70) {
                        myOutPut.innerHTML = "<h1>You got a D</h1>";
                        myOutPut.style.border = "2px solid orange";
                    } else if (score >= 0 && score < 60) {
                        myOutPut.innerHTML = "<h1>You got an F</h1>";
                        myOutPut.style.border = "2px solid maroon";
                    } else {
                        myOutPut.innerHTML = "<h1>Please enter a valid score</h1>";
                        myOutPut.style.border = "2px solid red";
                    }
                }
